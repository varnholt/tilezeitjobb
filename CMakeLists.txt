cmake_minimum_required(VERSION 3.24)
project(tilezeitjobb VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TZJ_BUILD_EDITOR "Build the ImGui editor" ON)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG d14d46f6b44456eb88beac58cba833a4135d4c89)
FetchContent_MakeAvailable(json stb)
add_library(tzj_core src/core/project.cpp src/core/image.cpp src/core/nes.cpp src/core/tools.cpp)
target_include_directories(tzj_core PUBLIC src PRIVATE ${stb_SOURCE_DIR})
target_link_libraries(tzj_core PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(tzj_core PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(tzj_core PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(tzj_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(tilezeitjobb-cli src/cli.cpp)
target_link_libraries(tilezeitjobb-cli PRIVATE tzj_core)
if(TZJ_BUILD_EDITOR)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG 52fe0a05a7b1aa180a202bb24f0f2a049a9c1b7d)
  FetchContent_MakeAvailable(glfw imgui)
  find_package(OpenGL REQUIRED)
  add_library(tzj_imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(tzj_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(tzj_imgui PUBLIC glfw OpenGL::GL)
  add_executable(tilezeitjobb src/app/main.cpp src/app/editor.cpp src/app/workflows.cpp)
  target_link_libraries(tilezeitjobb PRIVATE tzj_core tzj_imgui)
  if(WIN32)
    target_link_libraries(tilezeitjobb PRIVATE comdlg32)
    target_compile_definitions(tilezeitjobb PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  add_custom_command(TARGET tilezeitjobb POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/data/paltesteffect $<TARGET_FILE_DIR:tilezeitjobb>/data/paltesteffect)
  install(DIRECTORY data/paltesteffect DESTINATION bin/data)
  install(TARGETS tilezeitjobb RUNTIME DESTINATION bin)
  install(FILES ${imgui_SOURCE_DIR}/LICENSE.txt DESTINATION share/tilezeitjobb/licenses RENAME imgui-LICENSE.txt)
  install(FILES ${glfw_SOURCE_DIR}/LICENSE.md DESTINATION share/tilezeitjobb/licenses RENAME glfw-LICENSE.md)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(tzj_tests tests/core_tests.cpp)
  target_link_libraries(tzj_tests PRIVATE tzj_core)
  add_test(NAME core COMMAND tzj_tests)
  add_executable(tzj_workflow_tests tests/workflow_tests.cpp)
  target_link_libraries(tzj_workflow_tests PRIVATE tzj_core)
  add_test(NAME editing_workflows COMMAND tzj_workflow_tests)
endif()
install(TARGETS tilezeitjobb-cli RUNTIME DESTINATION bin)
install(FILES LICENSE README.md DESTINATION share/tilezeitjobb)

install(FILES ${json_SOURCE_DIR}/LICENSE.MIT DESTINATION share/tilezeitjobb/licenses RENAME json-LICENSE.MIT)
install(FILES ${stb_SOURCE_DIR}/LICENSE DESTINATION share/tilezeitjobb/licenses RENAME stb-LICENSE)
